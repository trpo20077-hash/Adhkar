extends CharacterBody3D @export var speed := 40.0 @export var pitch_speed := 1.2 @export var roll_speed := 2.0 @export var yaw_speed := 0.9 func _physics_process(delta: float) -> void: var pitch := Input.get_axis("ui_down", "ui_up") var roll := Input.get_axis("ui_right", "ui_left") rotate_object_local(Vector3.RIGHT, pitch * pitch_speed * delta) rotate_object_local(Vector3.FORWARD, roll * roll_speed * delta) rotate_object_local(Vector3.UP, -roll * yaw_speed * delta) velocity = -global_transform.basis.z * speed move_and_slide()
