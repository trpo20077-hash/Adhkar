extends CharacterBody3D var speed := 25.0 var pitch_speed := 1.2 var roll_speed := 2.0 var yaw_speed := 0.8 func _physics_process(delta): # فوق وتحت var pitch := Input.get_axis("ui_down", "ui_up") rotate_object_local(Vector3.RIGHT, pitch * pitch_speed * delta) # يمين ويسار: ميلان مع انعطاف var roll := Input.get_axis("ui_right", "ui_left") rotate_object_local(Vector3.FORWARD, roll * roll_speed * delta) rotate_y(roll * yaw_speed * delta) # الطيران للأمام velocity = -transform.basis.z * speed move_and_slide()
