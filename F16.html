extends CharacterBody3D @export var thrust := 50.0 @export var pitch_speed := 1.5 @export var roll_speed := 2.0 @export var yaw_speed := 1.0 var current_speed := 0.0 func _physics_process(delta: float) -> void: # زيادة تدريجية في الدفع للإقلاع current_speed = move_toward(current_speed, thrust, delta * 20.0) var pitch := Input.get_axis("ui_down", "ui_up") var roll := Input.get_axis("ui_right", "ui_left") rotate_object_local(Vector3.RIGHT, pitch * pitch_speed * delta) rotate_object_local(Vector3.FORWARD, roll * roll_speed * delta) rotate_object_local(Vector3.UP, -roll * yaw_speed * delta) # الحركة للأمام بناءً على اتجاه مقدمة الطائرة velocity = -global_transform.basis.z * current_speed # إضافة دفع بسيط للأعلى لمحاكاة قوة الرفع ومنع السقوط velocity.y += 9.8 * delta move_and_slide()
